cmake_minimum_required(VERSION 3.16)
project(vehicle_driver VERSION 1.0.0 LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
set(dependencies rclcpp rclcpp_components vehicle_core vehicle_interfaces geometry_msgs
  nav_msgs std_srvs diagnostic_msgs tf2_ros)
foreach(dependency IN LISTS dependencies)
  find_package(${dependency} REQUIRED)
endforeach()
add_library(vehicle_component SHARED src/vehicle_node.cpp)
target_compile_features(vehicle_component PUBLIC cxx_std_17)
target_compile_options(vehicle_component PRIVATE -Wall -Wextra -Wpedantic -Wshadow)
ament_target_dependencies(vehicle_component ${dependencies})
rclcpp_components_register_node(vehicle_component PLUGIN "vehicle_driver::VehicleNode"
  EXECUTABLE vehicle_node)
install(TARGETS vehicle_component ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(test_socketcan_integration test/test_integration.py TIMEOUT 60)
endif()
ament_package()
